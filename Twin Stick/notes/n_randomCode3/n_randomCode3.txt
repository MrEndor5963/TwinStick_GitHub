function enemy_damage_check(){
list_temp = ds_list_create() 
instance_place_list(x,y,[Bullet,PlayerWeapon,Explosion],list_temp,false)
var_repeat = 0

repeat(ds_list_size(list_temp)){
var_object = ds_list_find_value(list_temp,var_repeat)
if array_contains(contact_list,var_object) = false && var_object.hurts_enemy = true
{
	
	if var_object.object_index = PlayerWeapon{
	var_thing = instance_nearest(x,y,PlayerWeapon)
	direction = point_direction(x,y,var_thing.x,var_thing.y)+180
	speed = var_thing.knockback/enemy_weight
	hsp_knockback += hspeed
	vsp_knockback += vspeed
	speed = 0
	play_sfx(sfx_KnifeStab)}

	repeat(4){
	if var_object.object_index = PlayerWeapon or var_object.object_index = Explosion{particle = instance_create_depth(x,y,depth-100,ParticleEffect)}
	else{
	real_bullet_x = var_object.x + ((var_object.hspeed/var_object.speed)*(var_object.image_xscale*100))
	real_bullet_y = var_object.y + ((var_object.vspeed/var_object.speed)*(var_object.image_yscale*100))
	particle = instance_create_depth(real_bullet_x,real_bullet_y,depth-100,ParticleEffect)}
	particle.hsp = random_range(-12,12)
	particle.vsp = random_range(-18,4)
	particle.draw_color = blood_color
	}

	hit_stun = 2
	var_player = var_object.player_id
	bullet_reward = var_object.shot_reward
	if instance_exists(var_player) && var_player.object_index = Player{
	with var_player{player_point_change(other.bullet_reward)}}

	hp -= var_object.damage
	direction = var_object.direction
	speed = 1;
	creator.hsp_knockback = hspeed*var_object.knockback
	creator.vsp_knockback = vspeed*var_object.knockback
	speed = 0
	if var_object.object_index != Explosion && var_object.explosive = true{
	var_explosion = instance_create_depth(var_object.x,var_object.y,var_object.depth-1000,Explosion)
	var_explosion.creator = var_object.player_id
	var_explosion.damage = var_object.explosion_damage
	}
	
	if hp <= 0 or x < 0 or x> room_width or y < 0 or y > room_height{
	instance_destroy(creator);
	instance_destroy();
	ds_list_destroy(list_temp)
	if instance_exists(var_player) && var_player.object_index = Player{
	var_player.kills += 1;//throw("")
	if var_object.hurts_enemy = true{kill_reward = var_object.kill_reward}
	with var_player{player_point_change(other.kill_reward)}}
	var_object.penetration -= 1
	if var_object.penetration <= 0{instance_destroy(var_object)}
	blood_splatter()
	play_sfx(sfx_EnemyDeath)
	//freeze_frame(200)
	exit
	}

	if var_object.object_index = PlayerWeapon{array_push(var_object.contact_list,id)}
	var_object.penetration -= 1
	if var_object.penetration <= 0{if var_object.object_index = Bullet{instance_destroy(var_object)}}
	else{array_insert(contact_list,0,var_object)}
}

var_repeat += 1
}

ds_list_destroy(list_temp)
}
